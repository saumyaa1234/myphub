    <div class="bg-white rounded-lg p-6 w-full max-w-md">
        <h3 class="text-xl font-bold text-gray-800 mb-4">Edit Schedule Item</h3>
        <form id="edit-form" class="space-y-4">
            <input type="hidden" id="edit-id">
            <div>
                <label for="edit-title" class="block text-sm font-medium text-gray-700 mb-1">Title</label>
                <input type="text" id="edit-title" class="w-full px-4 py-2 border border-gray-300 rounded-md focus:outline-none focus:ring-2 focus:ring-blue-500" required>
            </div>
            <div>
                <label for="edit-location" class="block text-sm font-medium text-gray-700 mb-1">Location</label>
                <input type="text" id="edit-location" class="w-full px-4 py-2 border border-gray-300 rounded-md focus:outline-none focus:ring-2 focus:ring-blue-500">
            </div>
            <div class="flex space-x-2">
                <button type="submit" class="flex-1 bg-blue-600 hover:bg-blue-700 text-white font-bold py-2 px-4 rounded-md transition duration-300">Update</button>
                <button type="button" id="delete-item" class="bg-red-600 hover:bg-red-700 text-white font-bold py-2 px-4 rounded-md transition duration-300">Delete</button>
                <button type="button" id="cancel-edit" class="bg-gray-200 hover:bg-gray-300 text-gray-800 font-bold py-2 px-4 rounded-md transition duration-300">Cancel</button>

<script>
    // Global variables
    let scheduleItems = [];
    let selectedItemId = null;
    let nextItemId = 1;
    
    // Initialize when document is loaded
    document.addEventListener('DOMContentLoaded', function() {
        // Form submission for adding new items
        const scheduleForm = document.getElementById('schedule-form');
        scheduleForm.addEventListener('submit', function(e) {
            e.preventDefault();
            addScheduleItem();
        });
        
        // Clear schedule button
        document.getElementById('clear-schedule').addEventListener('click', function() {
            if (confirm('Are you sure you want to clear the entire schedule?')) {
                scheduleItems = [];
                renderSchedule();
            }
        });
        
        // Download button
        document.getElementById('download-btn').addEventListener('click', downloadSchedule);
        
        // Edit form submission
        document.getElementById('edit-form').addEventListener('submit', function(e) {
            e.preventDefault();
            updateScheduleItem();
        });
        
        // Delete button in edit modal
        document.getElementById('delete-item').addEventListener('click', function() {
            deleteScheduleItem(selectedItemId);
            closeEditModal();
        });
        
        // Cancel button in edit modal
        document.getElementById('cancel-edit').addEventListener('click', closeEditModal);
    });
    
    // Select color function
    function selectColor(element) {
        // Remove ring from all options
        document.querySelectorAll('.color-option').forEach(option => {
            option.classList.remove('ring-2', 'ring-offset-2');
        });
        
        // Add ring to selected option
        element.classList.add('ring-2', 'ring-offset-2');
        
        // Set the hidden input value
        document.getElementById('item-color').value = element.dataset.color;
    }
    
    // Add schedule item
    function addScheduleItem() {
        const title = document.getElementById('item-title').value;
        const location = document.getElementById('item-location').value;
        const day = document.getElementById('item-day').value;
        const startTime = document.getElementById('item-start').value;
        const endTime = document.getElementById('item-end').value;
        const colorClass = document.getElementById('item-color').value;
        
        // Validate end time is after start time
        const startHour = parseInt(startTime.split(':')[0]);
        const startMinute = parseInt(startTime.split(':')[1]);
        const endHour = parseInt(endTime.split(':')[0]);
        const endMinute = parseInt(endTime.split(':')[1]);
        
        if (endHour < startHour || (endHour === startHour && endMinute <= startMinute)) {
            alert('End time must be after start time');
            return;
        }
        
        // Create new item
        const newItem = {
            id: nextItemId++,
            title,
            location,
            day,
            startTime,
            endTime,
            colorClass
        };
        
        // Add to items array
        scheduleItems.push(newItem);
        
        // Reset form
        document.getElementById('schedule-form').reset();
        
        // Select first color by default
        document.querySelectorAll('.color-option').forEach((option, index) => {
            if (index === 0) {
                option.classList.add('ring-2', 'ring-offset-2');
            } else {
                option.classList.remove('ring-2', 'ring-offset-2');
            }
        });
        document.getElementById('item-color').value = 'bg-blue-500 text-white';
        
        // Render schedule
        renderSchedule();
    }
    
    // Render schedule
    function renderSchedule() {
        // Clear existing items
        document.querySelectorAll('.schedule-item').forEach(item => item.remove());
        
        // Render each item
        scheduleItems.forEach(item => {
            const dayColumn = document.getElementById(item.day);
            
            // Calculate position and height
            const startHour = parseInt(item.startTime.split(':')[0]);
            const startMinute = parseInt(item.startTime.split(':')[1]);
            const endHour = parseInt(item.endTime.split(':')[0]);
            const endMinute = parseInt(item.endTime.split(':')[1]);
            
            const startPosition = (startHour - 8) * 60 + startMinute;
            const endPosition = (endHour - 8) * 60 + endMinute;
            const height = endPosition - startPosition;
            
            // Create item element
            const itemElement = document.createElement('div');
            itemElement.className = `schedule-item ${item.colorClass}`;
            itemElement.style.top = `${startPosition}px`;
            itemElement.style.height = `${height}px`;
            itemElement.style.width = 'calc(100% - 8px)';
            itemElement.style.left = '4px';
            
            // Format time for display
            const displayStartTime = formatTime(item.startTime);
            const displayEndTime = formatTime(item.endTime);
            
            // Set content
            itemElement.innerHTML = `
                <div class="font-semibold">${item.title}</div>
                ${item.location ? `<div class="text-xs opacity-90">${item.location}</div>` : ''}
                <div class="text-xs mt-1">${displayStartTime} - ${displayEndTime}</div>
            `;
            
            // Add click event for editing
            itemElement.addEventListener('click', function() {
                openEditModal(item);
            });
            
            // Add to day column
            dayColumn.appendChild(itemElement);
        });
    }
    
    // Format time (convert 24h to 12h)
    function formatTime(time) {
        const [hour, minute] = time.split(':');
        const hourNum = parseInt(hour);
        
        if (hourNum < 12) {
            return `${hourNum === 0 ? 12 : hourNum}:${minute} AM`;
        } else {
            return `${hourNum === 12 ? 12 : hourNum - 12}:${minute} PM`;
        }
    }
    
    // Open edit modal
    function openEditModal(item) {
        selectedItemId = item.id;
        
        document.getElementById('edit-id').value = item.id;
        document.getElementById('edit-title').value = item.title;
        document.getElementById('edit-location').value = item.location;
        
        document.getElementById('edit-modal').classList.remove('hidden');
    }
    
    // Close edit modal
    function closeEditModal() {
        document.getElementById('edit-modal').classList.add('hidden');
        selectedItemId = null;
    }
    
    // Update schedule item
    function updateScheduleItem() {
        const id = parseInt(document.getElementById('edit-id').value);
        const title = document.getElementById('edit-title').value;
        const location = document.getElementById('edit-location').value;
        
        // Find and update item
        const itemIndex = scheduleItems.findIndex(item => item.id === id);
        if (itemIndex !== -1) {
            scheduleItems[itemIndex].title = title;
            scheduleItems[itemIndex].location = location;
            
            // Re-render schedule
            renderSchedule();
            
            // Close modal
            closeEditModal();
        }
    }
    
    // Delete schedule item
    function deleteScheduleItem(id) {
        scheduleItems = scheduleItems.filter(item => item.id !== id);
        renderSchedule();
    }
    
    // Download schedule as image
    function downloadSchedule() {
        const scheduleElement = document.getElementById('schedule');
        
        html2canvas(scheduleElement, {
            backgroundColor: '#ffffff',
            scale: 2 // Higher quality
        }).then(canvas => {
            // Create download link
            const link = document.createElement('a');
            link.download = 'my-schedule.png';
            link.href = canvas.toDataURL('image/png');
            link.click();
        });
    }
</script>
